```python
"""Full-bleed 1920×1080 LTR graph. Same drawing for Slide 1 and video Beat 3.

STEP rail | dashed capture boundary | SPAN | binding gap | CLAIM | gate | ACTION
Six span rows, six claim rows, nine step ticks with four dead.
14 chips = 4 rows x (source, ACL, hash) + billing's (ACL, CALLER).
Every container holds something: an empty box reads as unfinished, not as texture.
Two colours carry meaning: bound (neutral) and unbound (red).
"""

from .tokens import PLATE, PLATE_2, LINE, BOUND, RED, VIOLET, WHITE, MONO, DISPLAY

W, H = 1920, 1080
ROW0, ROW_H, SPAN_H = 170, 112, 92
BODY_BOT = ROW0 + 5 * ROW_H + SPAN_H              # 822

STEP_X, STEP_W = 56, 148
BOUND_X = 226
SPAN_X, SPAN_W = 248, 466
CLAIM_X, CLAIM_W = 850, 420
GATE_X = 1312
ACT_X, ACT_W = 1372, 492                          # right edge 1864

BAND_Y = 892
N_TICKS, TICK_H = 9, 48
DEAD = {2, 4, 6, 7}
DIM = 0.16
C_COST, C_RESP, C_PERF = "#7FBFA8", "#E0A44C", "#8FA0DC"

# name, kind: full | ghost | acl | empty
SPANS = [
    ("policy.db", "full"),
    ("tickets", "full"),
    ("contract", "ghost"),
    ("ledger", "ghost"),
    ("billing", "acl"),
    (None, "empty"),
]
CLAIMS = ["vendor id", "order date", "amount paid", "return window", "account"]


def _row_y(i):
    return ROW0 + i * ROW_H


def _chips(x, y, labels, ghost=False):
    cw, ch, gap = 134, 30, 10
    ink = "#5A5C64" if ghost else "#D0CED6"
    fill = "#14161B" if ghost else "#1C1E24"
    stroke = "#24262C" if ghost else "#2E3038"
    out = []
    for i, label in enumerate(labels):
        bx = x + i * (cw + gap)
        out.append(
            f'<rect x="{bx}" y="{y}" width="{cw}" height="{ch}" fill="{fill}" stroke="{stroke}"/>'
            f'<text x="{bx + cw / 2:.1f}" y="{y + 21}" font-family="{MONO}" font-size="13" '
            f'fill="{ink}" text-anchor="middle">{label}</text>'
        )
    return "".join(out)


def graph_svg(
    bound=5,
    show_claims=True,
    show_action=True,
    show_spans=True,
    show_gate=True,
    highlight=None,
    show_leaders=True,
    show_cost_count=False,
    chrome=False,
):
    dim_step = highlight in ("perf", "resp")
    dim_span = highlight in ("perf", "cost")
    dim_claim = highlight in ("resp", "cost")
    dim_act = highlight in ("perf", "resp", "cost")

    p = [
        f'<svg viewBox="0 0 {W} {H}" width="100%" height="100%" xmlns="http://www.w3.org/2000/svg">',
        f"<rect width='{W}' height='{H}' fill='{PLATE}'/>",
    ]

    if chrome:
        p.append(
            f'<text x="56" y="44" font-family="{DISPLAY}" font-size="16" font-style="italic" '
            f'fill="#8A8790">It used to be a bad paragraph. It is now an executed transaction.</text>'
            f'<text x="56" y="94" font-family="{DISPLAY}" font-size="36" font-weight="700" '
            f'fill="{WHITE}" letter-spacing="-0.8">An AI response is a set of claims requesting '
            f'permission to act.</text>'
        )

    def lab(x, title, sub):
        return (
            f'<text x="{x}" y="132" font-family="{DISPLAY}" font-size="13" font-weight="700" '
            f'letter-spacing="3" fill="#C8C6CE">{title}</text>'
            f'<text x="{x}" y="153" font-family="{MONO}" font-size="13" fill="#8A8790">{sub}</text>'
        )

    p.append(lab(STEP_X, "STEP", "tool · retrieval"))
    p.append(lab(SPAN_X, "SPAN", "source · ACL · hash"))
    p.append(lab(CLAIM_X, "CLAIM", "default: UNSUPPORTED"))
    p.append(lab(ACT_X, "ACTION", "irreversible · R3"))
    p.append(
        f'<rect x="{SPAN_X + 92}" y="118" width="30" height="18" fill="{VIOLET}"/>'
        f'<text x="{SPAN_X + 107}" y="132" font-family="{MONO}" font-size="12" font-weight="700" '
        f'fill="#100E14" text-anchor="middle">14</text>'
    )
    for x in (STEP_X + 152, CLAIM_X - 52, ACT_X - 48):
        p.append(
            f'<path d="M {x} 126 H {x+30}" stroke="#5A5C64" stroke-width="1.6"/>'
            f'<path d="M {x+22} 120 L {x+32} 126 L {x+22} 132" fill="none" stroke="#5A5C64" '
            f'stroke-width="1.6"/>'
        )

    p.append(
        f'<line x1="{BOUND_X}" y1="{ROW0 - 6}" x2="{BOUND_X}" y2="{BODY_BOT}" stroke="{VIOLET}" '
        f'stroke-width="2" stroke-dasharray="6 8"/>'
        f'<text x="{BOUND_X + 12}" y="{ROW0 - 14}" font-family="{DISPLAY}" font-size="13" '
        f'font-weight="700" fill="{VIOLET}">Captured here — outside the model</text>'
    )

    # ---- STEP rail
    top = DIM if dim_step else 1
    tstep = (BODY_BOT - ROW0 - TICK_H) / (N_TICKS - 1)
    for i in range(N_TICKS):
        y = ROW0 + i * tstep
        dead = i in DEAD
        p.append(
            f'<rect x="{STEP_X}" y="{y:.1f}" width="{STEP_W}" height="{TICK_H}" '
            f'fill="{"#22242B" if dead else "#D4D4DC"}" opacity="{top}"/>'
        )
        if dead:
            p.append(
                f'<line x1="{STEP_X + 10}" y1="{y + TICK_H - 9:.1f}" x2="{STEP_X + STEP_W - 10}" '
                f'y2="{y + 9:.1f}" stroke="#4A4D57" stroke-width="2" opacity="{top}"/>'
            )

    if show_leaders and highlight in (None, "cost", "all"):
        ay = BODY_BOT + 20
        p.append(
            f'<path d="M {STEP_X + STEP_W} {ay} H {STEP_X + 14}" stroke="{C_COST}" stroke-width="2.4"/>'
            f'<path d="M {STEP_X + 27} {ay - 7} L {STEP_X + 12} {ay} L {STEP_X + 27} {ay + 7}" '
            f'fill="none" stroke="{C_COST}" stroke-width="2.4"/>'
            f'<text x="{STEP_X}" y="{BODY_BOT + 44}" font-family="{MONO}" font-size="13" '
            f'fill="#8A8790">4 of 9 grounded nothing</text>'
        )

    # ---- SPAN column
    if show_spans:
        for i, (name, kind) in enumerate(SPANS):
            y = _row_y(i)
            d = dim_span and not (highlight == "resp" and kind == "acl")
            op = DIM if d else 1
            if kind == "empty":
                p.append(
                    f'<g opacity="{0.5 if not d else 0.12}">'
                    f'<line x1="{SPAN_X}" y1="{y + SPAN_H / 2}" x2="{SPAN_X + SPAN_W}" '
                    f'y2="{y + SPAN_H / 2}" stroke="#2A2D34" stroke-width="1"/></g>'
                )
                continue
            if kind == "acl":
                p.append(
                    f'<g opacity="{op}">'
                    f'<rect x="{SPAN_X}" y="{y}" width="{SPAN_W}" height="{SPAN_H}" fill="#1A1214" '
                    f'stroke="{RED}" stroke-width="2.4"/>'
                    f'<text x="{SPAN_X+18}" y="{y+28}" font-family="{MONO}" font-size="17" '
                    f'font-weight="700" fill="{RED}">{name}</text>'
                    f'<rect x="{SPAN_X+18}" y="{y+44}" width="182" height="32" fill="#2C1A1D"/>'
                    f'<text x="{SPAN_X+28}" y="{y+66}" font-family="{MONO}" font-size="14" '
                    f'font-weight="700" fill="{WHITE}">ACL  risk-team</text>'
                    f'<text x="{SPAN_X+212}" y="{y+68}" font-family="{DISPLAY}" font-size="22" '
                    f'font-weight="700" fill="{RED}">≠</text>'
                    f'<rect x="{SPAN_X+234}" y="{y+44}" width="196" height="32" fill="#2C1A1D"/>'
                    f'<text x="{SPAN_X+244}" y="{y+66}" font-family="{MONO}" font-size="14" '
                    f'font-weight="700" fill="{WHITE}">CALLER  customer</text>'
                    f"</g>"
                )
                continue
            ghost = kind == "ghost"
            p.append(
                f'<g opacity="{op}">'
                f'<rect x="{SPAN_X}" y="{y}" width="{SPAN_W}" height="{SPAN_H}" fill="{PLATE_2}" '
                f'stroke="{LINE}"/>'
                f'<text x="{SPAN_X+18}" y="{y+30}" font-family="{MONO}" font-size="17" '
                f'font-weight="600" fill="{"#8C8A93" if ghost else "#CFCDD4"}">{name}</text>'
                f'{_chips(SPAN_X + 18, y + 46, ["source", "ACL", "hash"], ghost=ghost)}'
                f"</g>"
            )

    # ---- bindings
    if show_claims and bound > 0:
        for i in range(min(bound, 5)):
            y = _row_y(i) + SPAN_H / 2
            d = dim_claim or dim_span
            if highlight == "resp" and i == 4:
                d = False
            if highlight == "perf":
                d = True
            col = RED if i == 4 else "#6E7079"
            o = DIM if d else 1
            p.append(
                f'<line x1="{SPAN_X + SPAN_W}" y1="{y}" x2="{CLAIM_X}" y2="{y}" stroke="{col}" '
                f'stroke-width="{2.6 if i == 4 else 1.8}" opacity="{o}"/>'
                f'<circle cx="{SPAN_X + SPAN_W + 6}" cy="{y}" r="4" fill="{col}" opacity="{o}"/>'
                f'<circle cx="{CLAIM_X - 6}" cy="{y}" r="4" fill="{col}" opacity="{o}"/>'
            )

    # ---- CLAIM column
    if show_claims:
        for i in range(6):
            y = _row_y(i)
            cy = y + SPAN_H / 2
            is_red = i >= bound
            d = dim_claim and not (highlight == "perf" and is_red)
            if is_red and highlight in ("resp", "cost"):
                d = True
            op = DIM if d else 1
            if is_red:
                p.append(
                    f'<g opacity="{op}">'
                    f'<rect x="{CLAIM_X}" y="{y}" width="{CLAIM_W}" height="{SPAN_H}" '
                    f'fill="#1A1214" stroke="{RED}" stroke-width="3"/>'
                    f'<text x="{CLAIM_X+20}" y="{y+40}" font-family="{MONO}" font-size="18" '
                    f'font-weight="700" fill="{RED}">clause 7.2</text>'
                    f'<text x="{CLAIM_X+20}" y="{y+68}" font-family="{MONO}" font-size="15" '
                    f'fill="{RED}">no span — UNSUPPORTED</text>'
                    f"</g>"
                )
            else:
                p.append(
                    f'<g opacity="{op}">'
                    f'<rect x="{CLAIM_X}" y="{y}" width="{CLAIM_W}" height="{SPAN_H}" '
                    f'fill="{PLATE_2}" stroke="{LINE}"/>'
                    f'<circle cx="{CLAIM_X+28}" cy="{cy}" r="7" fill="{BOUND}"/>'
                    f'<text x="{CLAIM_X+52}" y="{cy+6}" font-family="{MONO}" font-size="17" '
                    f'fill="#CFCDD4">{CLAIMS[i]}</text>'
                    f"</g>"
                )

    # ---- severed edge
    if show_claims and bound >= 5 and show_gate:
        cy = _row_y(5) + SPAN_H / 2
        op = DIM if highlight in ("resp", "cost") else 1
        p.append(
            f'<g opacity="{op}">'
            f'<line x1="{CLAIM_X + CLAIM_W}" y1="{cy}" x2="{GATE_X - 18}" y2="{cy}" stroke="{RED}" '
            f'stroke-width="2.6"/>'
            f'<line x1="{GATE_X-16}" y1="{cy-16}" x2="{GATE_X+16}" y2="{cy+16}" stroke="{RED}" '
            f'stroke-width="4"/>'
            f'<line x1="{GATE_X+16}" y1="{cy-16}" x2="{GATE_X-16}" y2="{cy+16}" stroke="{RED}" '
            f'stroke-width="4"/></g>'
        )

    # ---- ACTION: one plate
    if show_action:
        op = 0.22 if dim_act else 1
        mid = (ROW0 + BODY_BOT) / 2
        p.append(
            f'<g opacity="{op}">'
            f'<rect x="{ACT_X}" y="{ROW0}" width="{ACT_W}" height="{BODY_BOT - ROW0}" '
            f'fill="#14161B" stroke="{WHITE}" stroke-width="2"/>'
            f'<text x="{ACT_X + 36}" y="{mid - 20}" font-family="{DISPLAY}" font-size="58" '
            f'font-weight="700" fill="{WHITE}">₹1,84,000</text>'
            f'<text x="{ACT_X + 36}" y="{mid + 16}" font-family="{MONO}" font-size="15" '
            f'fill="#8A8790">refund · vendor agreement</text>'
            f'<rect x="{ACT_X + 36}" y="{mid + 46}" width="248" height="42" fill="#2C1A1D" '
            f'stroke="{RED}" stroke-width="1.6"/>'
            f'<text x="{ACT_X + 54}" y="{mid + 73}" font-family="{MONO}" font-size="16" '
            f'font-weight="700" fill="{RED}">HELD — not executed</text>'
            f"</g>"
        )

    # ---- three reads, one band, under the column each reads
    if show_leaders:
        def leader(x, axis, line, on, col):
            o = 1 if on else 0.14
            return (
                f'<g opacity="{o}">'
                f'<rect x="{x}" y="{BAND_Y}" width="4" height="46" fill="{col}"/>'
                f'<text x="{x+16}" y="{BAND_Y+17}" font-family="{DISPLAY}" font-size="12" '
                f'font-weight="700" letter-spacing="2.4" fill="{col}">{axis}</text>'
                f'<text x="{x+16}" y="{BAND_Y+43}" font-family="{DISPLAY}" font-size="22" '
                f'font-weight="700" fill="{WHITE}">{line}</text></g>'
            )

        p.append(leader(STEP_X, "COST", "Unused Step",
                        highlight in (None, "cost", "all"), C_COST))
        p.append(leader(SPAN_X, "RESPONSIBILITY", "Unentitled Span",
                        highlight in (None, "resp", "all"), C_RESP))
        p.append(leader(CLAIM_X, "PERFORMANCE", "Unbound Claim",
                        highlight in (None, "perf", "all"), C_PERF))

    if chrome:
        p.append(
            f'<text x="56" y="1006" font-family="{DISPLAY}" font-size="22" font-weight="700" '
            f'fill="{WHITE}">Three dimensions, one graph.</text>'
            f'<text x="56" y="1034" font-family="{DISPLAY}" font-size="15" fill="#8A8790">'
            f"The system didn’t fail. It was never asked to prove anything.</text>"
        )

    p.append("</svg>")
    return "".join(p)
```
